cmake_minimum_required(VERSION 3.21)
project(DiTunnelShellIcon C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The taskbar component requires Windows x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  string(REPLACE "/O2" "" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
endif()
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
add_library(DiTunnel.ShellIcon SHARED shell_icon.c)
target_link_libraries(DiTunnel.ShellIcon PRIVATE minhook user32 psapi)
if(MSVC)
  target_compile_options(minhook PRIVATE $<$<CONFIG:Release>:/O2>)
  target_compile_options(DiTunnel.ShellIcon PRIVATE $<$<CONFIG:Release>:/O2>)
  target_compile_options(DiTunnel.ShellIcon PRIVATE /W4 /guard:cf)
  target_link_options(DiTunnel.ShellIcon PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
endif()
option(DITUNNEL_SHELL_TESTS "Build tests which never attach to Explorer" OFF)
if(DITUNNEL_SHELL_TESTS)
  add_library(DiTunnel.ShellIcon.Test SHARED shell_icon.c)
  target_compile_definitions(DiTunnel.ShellIcon.Test PRIVATE DITUNNEL_SHELL_TESTS)
  target_compile_options(DiTunnel.ShellIcon.Test PRIVATE /Od /W4)
  target_link_libraries(DiTunnel.ShellIcon.Test PRIVATE minhook user32 psapi)
  add_executable(DiTunnel.ShellIcon.SelfTest self_test.c)
  target_link_libraries(DiTunnel.ShellIcon.SelfTest PRIVATE user32)
endif()
